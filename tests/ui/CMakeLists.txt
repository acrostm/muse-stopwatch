cmake_minimum_required(VERSION 3.20)
project(stopwatch_ui_test C CXX ASM)
if(NOT LVGL_SOURCE_DIR)
    message(FATAL_ERROR "Set LVGL_SOURCE_DIR to the pinned SDK managed LVGL source")
endif()
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
add_link_options(-fsanitize=address,undefined)
add_subdirectory("${LVGL_SOURCE_DIR}" lvgl EXCLUDE_FROM_ALL)
add_executable(test_pages test_pages.c ../../src/sw_pages.c ../../src/sw_core.c)
target_include_directories(test_pages BEFORE PRIVATE stubs ../../src)
target_compile_definitions(test_pages PRIVATE muse_font_cjk_16=lv_font_montserrat_16)
target_compile_options(test_pages PRIVATE -Wall -Wextra -Werror)
target_link_libraries(test_pages PRIVATE lvgl m)
